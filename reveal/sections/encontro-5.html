10 / 15
Tool + Executor
Agir com segurança

A tool calcula. O executor decide o que pode rodar.

tool local previsívelallowlist explícita
def planejar_blocos(minutos):
    quantidade = max(1, min(3, minutos // 10))
    return f"{quantidade} bloco(s)"

TOOLS = {"planejar_blocos": planejar_blocos}

def executar_tool(nome, argumento):
    if nome not in TOOLS:
        raise ValueError("Tool não permitida")
    return TOOLS[nome](argumento)
